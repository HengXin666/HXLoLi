写到下面去就没问题了 =-=

#include <WinSock2.h>
#include <MSWSock.h>

#include <Windows.h>

#pragma comment(lib, "Ws2_32.lib")
#pragma comment(lib, "Mswsock.lib")

// --- 真下头 ---
#include <graphics.h>
#include <WinUser.h>
C++
